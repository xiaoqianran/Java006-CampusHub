# 后端

### 启动 user 服务（前台）
java -jar shiqian-user/target/shiqian-user-1.0.0-SNAPSHOT.jar --spring.profiles.active=local --spring.cloud.nacos.discovery.enabled=true

### 启动 resource 服务（前台）

### 启动 gateway 服务（前台）
java -jar shiqian-gateway/target/shiqian-gateway-1.0.0-SNAPSHOT.jar --spring.profiles.active=local


## 启动 User 服务（8081 端口）

nohup java -jar shiqian-user/target/shiqian-user-1.0.0-SNAPSHOT.jar \
  --spring.profiles.active=local --spring.cloud.nacos.discovery.enabled=true \
  > logs/fresh-user.log 2>&1 &

## 启动 Resource 服务（8082 端口）
nohup java -jar shiqian-resource/target/shiqian-resource-1.0.0-SNAPSHOT.jar \
  --spring.profiles.active=local --spring.cloud.nacos.discovery.enabled=true \
  > logs/fresh-resource.log 2>&1 &

## 启动 Gateway 服务（8080 端口）
nohup java -jar shiqian-gateway/target/shiqian-gateway-1.0.0-SNAPSHOT.jar \
  --spring.profiles.active=local \
  > logs/fresh-gateway.log 2>&1 &

echo "✅ 后端三个服务已发送启动指令"
echo "请等待 15~25 秒后执行检查命令"
